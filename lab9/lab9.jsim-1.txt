* FULL ADDER AND 32-BIT RIPPLE ADDER
.subckt fa a b c_in sum c_out
  * Sum
  Xsum1 a b n1 xor2
  Xsum2 c_in n1 sum xor2

  * Carry
  Xc1 a b n2 nand2
  Xc2 c_in n1 n3 nand2
  Xc3 n2 n3 c_out nand2 
.ends

.subckt add32 a[31:0] b[31:0] cin0 sum[31:0] cout[31:30]
  Xfa a[31:0] b[31:0] cout[30:0] cin0 sum[31:0] cout[31:0] fa
.ends

* 32-BIT ADDER/SUBTRACTOR  
.subckt as32 a[31:0] b[31:0] op0 fas[31:0] n v z 
    * Interpret the opcode 
    Xinvb b[31:0] invb[31:0] inverter
    Xmux2 op0#32 b[31:0] invb[31:0] b_mux[31:0] mux2
    Xadd32 a[31:0] b_mux[31:0] op0 fas[31:0] cout[31:30] add32 
    
    * === Flags for Negative, check last bit=== 
    .connect fas31 n 

    * === Flags for Overflow, last two bits cannot be same====  
    Xv cout31 cout30 v xor2 

    * === Flags for Zero === 
    Xz1 fas[31:24] fas[23:16] fas[15:8] fas[7:0] nor_out[7:0] nor4
    Xz2 nor_out[7:4] nor_out[3:0] nand_out[1:0] nand4
    Xz3 nand_out[1:0] z nor2  
.ends

* 32-BIT COMPARATOR
.subckt cmp32 n v z op[2:1] fcmp[31:0] 
    Xequal z op1 equal nand2
    Xxor n v nxorv xor2
    Xless op2 nxorv less nand2
    Xlesseq equal less fcmp0 nand2
    .connect 0 fcmp[31:1]
.ends 

* 32-BIT BOOLEAN  
.subckt bool32 a[31:0] b[31:0] op[3:0] fbool[31:0] 
    Xmux4 a[31:0] b[31:0] op0#32 op1#32 op2#32 op3#32 fbool[31:0] mux4
.ends 

* 32-BIT SHIFT SUBCIRCUIT 
.subckt shift32 a[31:0] b[4:0] op[1:0] fshift[31:0]
    XmuxLorR op0#32 a[0:31] a[31:0] val[31:0] mux2
    Xsign op1 a31 f and2

    Xmux2shift16 b4#32 val[31:0] f#16 val[31:16] shifted16_[31:0] mux2
    Xmux2shift8 b3#32 shifted16_[31:0] f#8 shifted16_[31:8] shifted8_[31:0] mux2
    Xmux2shift4 b2#32 shifted8_[31:0] f#4 shifted8_[31:4] shifted4_[31:0] mux2
    Xmux2shift2 b1#32 shifted4_[31:0] f#2 shifted4_[31:2] shifted2_[31:0] mux2
    Xmux2shift1 b0#32 shifted2_[31:0] f#1 shifted2_[31:1] shift[31:0] mux2

    Xmux2 op0#32 shift[0:31] shift[31:0] fshift[31:0] mux2
.ends

* 32-BIT ARITHEMTIC LOGIC UNIT SUBCIRCUIT 
.subckt alu32 a[31:0] b[31:0] alufn[5:0] f[31:0]
    Xas32 a[31:0] b[31:0] alufn0 as_out[31:0] n v z as32
    Xshift32 a[31:0] b[4:0] alufn[1:0] shift_out[31:0] shift32
    Xbool32 a[31:0] b[31:0] alufn[3:0] bool_out[31:0] bool32
    Xcmp32 n v z alufn[2:1] cmp_out[31:0] cmp32
    * 100 000  
    Xselect alufn5#32 alufn4#32 as_out[31:0] shift_out[31:0] bool_out[31:0] cmp_out[31:0] f[31:0] mux4
.ends

*******************************************************************************
* Xmemory is a 512-byte main memory containing both user instructions and data.
*   1st line = read port for the next instruction to process (inst)
*   2nd line = read port for LD instructions (memrd)
*   3rd line = write port for ST instructions (memwd, controlled by wr signal)
*   5th line = the memory statement itself

+        vdd  0    0   mema[8:2]  memrd[31:0]
+        0    clk  wr  mema[8:2]  memwd[31:0]
+ $memory width=32 nlocations=128 file="lab7.bin"
